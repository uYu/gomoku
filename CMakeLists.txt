cmake_minimum_required(VERSION 3.18.2)
project(gomoku_atelier VERSION 2.0.0 LANGUAGES C CXX)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE ATELIER_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(EXISTS "${ATELIER_SDK}/usr/include/c++/v1")
    include_directories(SYSTEM "${ATELIER_SDK}/usr/include/c++/v1")
  endif()
endif()
set(NO_COMMAND_MODULES OFF CACHE BOOL "Include training commands")
set(NO_MULTI_THREADING OFF CACHE BOOL "Native engine supports threads")
add_subdirectory(third_party/rapfi/Rapfi)
set_target_properties(rapfi PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

foreach(bridge rules dataset)
 add_library(gomoku_${bridge} SHARED cpp/${bridge}.cpp
  third_party/rapfi/Rapfi/game/board.cpp third_party/rapfi/Rapfi/game/pattern.cpp
  third_party/rapfi/Rapfi/core/hash.cpp)
 target_include_directories(gomoku_${bridge} PRIVATE third_party/rapfi/Rapfi/external/lz4/include third_party/rapfi/Rapfi/external/cpptoml/include)
 set_target_properties(gomoku_${bridge} PROPERTIES CXX_STANDARD 17 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
 if(APPLE)
  target_link_options(gomoku_${bridge} PRIVATE -Wl,-dead_strip)
 elseif(UNIX)
  target_compile_options(gomoku_${bridge} PRIVATE -ffunction-sections -fdata-sections)
  target_link_options(gomoku_${bridge} PRIVATE -Wl,--gc-sections)
 endif()
endforeach()
